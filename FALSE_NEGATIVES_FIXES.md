### **1. Missing Data Minimization in Consent Storage**
**Frameworks:** Belmont Report (Respect for Persons), GDPR (Article 5)

**Issue:** Consent state was stored with potentially user-identifiable session IDs.

**Fix:** `consent_manager.js`
- ✅ **Anonymous Session IDs:** Uses cryptographically secure random IDs (no user-identifiable info)
- ✅ **Minimal Data Storage:** Only stores necessary consent data (ai, tools, timestamp)
- ✅ **Hex-only Format:** Session IDs use only hex characters (0-9, a-f) to prevent encoding user data
- ✅ **Fixed Length:** 64-character limit for data minimization
- ✅ **Automatic Cleanup:** 24-hour TTL for all session data

**Code:**
```javascript
// Generate anonymous session ID
function generateAnonymousSessionId() {
  return crypto.randomBytes(32).toString('hex').substring(0, 64);
}

// Validate session ID format
function isValidSessionId(sessionId) {
  if (typeof sessionId !== 'string') return false;
  if (sessionId.length === 0 || sessionId.length > 64) return false;
  return /^[a-f0-9]+$/.test(sessionId); // Hex only
}
```

const CRISIS_ESCALATION_LEVELS = Object.freeze({

function determineCrisisEscalation(input, sensitivity) {
  if (sensitivity.category === 'crisis' && sensitivity.severity === 'imminent') {
    return CRISIS_ESCALATION_LEVELS.IMMEDIATE;
  }
  // ... other conditions
}

function generateCrisisResponseWithEscalation(input, sensitivity, formatHumanNLP, truncateForMirror) {
  const escalation = determineCrisisEscalation(input, sensitivity);
  return formatHumanNLP({
    userInput: input,
    anchor: escalation.message.split('.')[0] + '.',
    mirror: `You shared: "${truncateForMirror(input, 50)}"`,
    reframe: escalation.message,
    rapport: escalation.humanSupport

      : `Would you like me to share more resources?`,
    isCrisis: true,
    escalationLevel: escalation.name,
    requiresHumanSupport: escalation.humanSupport,
  });
}

function simplifyText(text) {
  // Replace complex words with simpler alternatives
  const simplifications = { 'utilize': 'use', 'individual': 'person', ... };
  let simplified = text;
  for (const [complex, simple] of Object.entries(simplifications)) {

- ✅ **Cleanup:** `clearAuditLog(sessionId)` - Clear logs for compliance
- ✅ **Anonymous IDs:** Only stores anonymous session IDs

**Code:**
```javascript
const AUDIT_LOG = [];

function logEthicalDecision(event, details = {}, sessionId = null) {
  const normalizedSessionId = normalizeSessionId(sessionId);
  const auditEntry = {
    timestamp: new Date().toISOString(),
    event,
    details,
    sessionId: normalizedSessionId,
  };
  AUDIT_LOG.push(auditEntry);
  console.log(`[ETHICS_AUDIT] ${event}:`, auditEntry);
}

function getAuditLog(sessionId = null) {
  const normalizedSessionId = normalizeSessionId(sessionId);
  return normalizedSessionId
    ? AUDIT_LOG.filter(entry => entry.sessionId === normalizedSessionId)
    : [...AUDIT_LOG];
}
```
